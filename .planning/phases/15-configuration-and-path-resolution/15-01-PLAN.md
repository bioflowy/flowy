```go
func cleanPath(p string, wdlDir string) string {
    if !filepath.IsAbs(p) {
        if wdlDir != "" {
            return filepath.Clean(filepath.Join(wdlDir, p))
        }
        if abs, err := filepath.Abs(p); err == nil {
            return filepath.Clean(abs)
        }
    }
    return filepath.Clean(p)
}
```
