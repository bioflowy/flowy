       Algorithm:
       ```
       running := map[string]bool{}
       for _, id := range order {
           node := nodes[id]
           // Wait until all deps done
           for !allDepsDone(node) {
               msg := <-resultCh  // blocks
               handleCompletion(msg)
           }
           // Dispatch node
           if isSync(node) { // Decl
               handleDecl(node)
           } else {
               running[id] = true
               g.Go(dispatch node -> resultCh)
           }
       }
       // Drain remaining running nodes
       for len(running) > 0 {
           msg := <-resultCh
           handleCompletion(msg)
       }
       ```

      ```go
      func (s *scheduler) runScatter(ctx context.Context, g *errgroup.Group,
          scatter *ast.Scatter, env *eval.Env) (map[string]values.Value, error) {

          arrVal, err := eval.Eval(scatter.Expr, env, eval.ExecContext{})
          if err != nil { return nil, err }
          arr := arrVal.(values.ArrayValue)

          if len(arr.Items) == 0 {
              // Return empty arrays for each call output
              return emptyScatterOutputs(scatter.Body, s.taskIndex), nil
          }

          type elemResult struct {
              idx     int
              outputs map[string]values.Value
              err     error
          }
          elemCh := make(chan elemResult, len(arr.Items))

          for i, item := range arr.Items {
              i, item := i, item
              g.Go(func() error {
                  scatterEnv := eval.NewEnv(env)
                  scatterEnv.Define(scatter.Variable, item)
                  outputs, err := s.executeBody(ctx, scatter.Body, scatterEnv)
                  elemCh <- elemResult{idx: i, outputs: outputs, err: err}
                  if err != nil { return err }
                  return nil
              })
          }

          // Collect all results
          results := make([]map[string]values.Value, len(arr.Items))
          for range arr.Items {
              select {
              case r := <-elemCh:
                  if r.err != nil { return nil, r.err }
                  results[r.idx] = r.outputs
              case <-ctx.Done():
                  return nil, ctx.Err()
              }
          }

          // Transpose: per-call output name → Array of values
          return transposeScatterResults(results, scatter.Body, s.taskIndex), nil
      }
      ```
