```typescript
import { useQuery } from '@tanstack/react-query'
import { apiClient } from '@/lib/apiClient'
import type { RunListItem } from '@/features/runs/useRuns'
import { TERMINAL_STATES } from '@/features/runs/useRuns'

export function usePollingRuns(slug: string) {
  return useQuery<RunListItem[]>({
    queryKey: ['projects', slug, 'runs'],
    queryFn: () => apiClient.get(`/projects/${slug}/runs`),
    enabled: !!slug,
    refetchInterval: (query) => {
      const data = query.state.data
      if (!data || data.length === 0) return false
      return data.some((r) => !TERMINAL_STATES.has(r.state)) ? 3000 : false
    },
    refetchIntervalInBackground: false,
  })
}
```

Key points:
- Uses same `queryKey` as `useRuns` so they share cache (React Query deduplication)
- `refetchInterval` callback receives the `Query` object (TanStack Query v5 signature), NOT data directly — access via `query.state.data`
- Returns `false` when data is undefined/empty (prevents polling for already-completed runs)
- Returns `3000` (3 seconds) when any run has a non-terminal state
- `refetchIntervalInBackground: false` pauses polling when browser tab is hidden

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'

// Mock @tanstack/react-query
const mockUseQuery = vi.fn()
vi.mock('@tanstack/react-query', () => ({
  useQuery: (...args: unknown[]) => {
    mockUseQuery(...args)
    return { data: undefined, isLoading: true, isError: false }
  },
}))

// Mock apiClient
vi.mock('@/lib/apiClient', () => ({
  apiClient: { get: vi.fn() },
}))

import { usePollingRuns } from '@/features/runs/usePollingRuns'
import { TERMINAL_STATES } from '@/features/runs/useRuns'

describe('TERMINAL_STATES', () => {
  it('contains exactly COMPLETED, FAILED, ABORTED', () => {
    expect(TERMINAL_STATES.has('COMPLETED')).toBe(true)
    expect(TERMINAL_STATES.has('FAILED')).toBe(true)
    expect(TERMINAL_STATES.has('ABORTED')).toBe(true)
    expect(TERMINAL_STATES.size).toBe(3)
  })
})

describe('usePollingRuns', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  function getRefetchInterval(): (query: { state: { data: unknown } }) => number | false {
    usePollingRuns('test-slug')
    const options = mockUseQuery.mock.calls[0][0]
    return options.refetchInterval
  }

  it('returns 3000 when any run is RUNNING', () => {
    const fn = getRefetchInterval()
    const result = fn({
      state: { data: [{ state: 'COMPLETED' }, { state: 'RUNNING' }] },
    })
    expect(result).toBe(3000)
  })

  it('returns 3000 when any run is PENDING', () => {
    const fn = getRefetchInterval()
    const result = fn({
      state: { data: [{ state: 'PENDING' }] },
    })
    expect(result).toBe(3000)
  })

  it('returns false when all runs are COMPLETED', () => {
    const fn = getRefetchInterval()
    const result = fn({
      state: { data: [{ state: 'COMPLETED' }, { state: 'COMPLETED' }] },
    })
    expect(result).toBe(false)
  })

  it('returns false when all runs are terminal (mixed COMPLETED and FAILED)', () => {
    const fn = getRefetchInterval()
    const result = fn({
      state: { data: [{ state: 'COMPLETED' }, { state: 'FAILED' }, { state: 'ABORTED' }] },
    })
    expect(result).toBe(false)
  })

  it('returns false when data is undefined', () => {
    const fn = getRefetchInterval()
    const result = fn({ state: { data: undefined } })
    expect(result).toBe(false)
  })

  it('returns false when data is empty array', () => {
    const fn = getRefetchInterval()
    const result = fn({ state: { data: [] } })
    expect(result).toBe(false)
  })

  it('passes refetchIntervalInBackground: false', () => {
    usePollingRuns('test-slug')
    const options = mockUseQuery.mock.calls[0][0]
    expect(options.refetchIntervalInBackground).toBe(false)
  })
})
```

This test strategy avoids fake timer issues (Pitfall 5 from RESEARCH.md) by testing the refetchInterval callback function directly.
