```go
package eval

import "github.com/bioflowy/flowy/wdl/values"

type Env struct {
    bindings map[string]values.Value
    parent   *Env
}

func NewEnv(parent *Env) *Env {
    return &Env{bindings: make(map[string]values.Value), parent: parent}
}

func (e *Env) Define(name string, v values.Value) {
    e.bindings[name] = v
}

func (e *Env) Lookup(name string) (values.Value, bool) {
    for cur := e; cur != nil; cur = cur.parent {
        if v, ok := cur.bindings[name]; ok {
            return v, true
        }
    }
    return nil, false
}
```

**wdl/eval/stdlib.go** — stdlib dispatch infrastructure + evalApply stub:
```go
package eval

import "github.com/bioflowy/flowy/wdl/values"

type stdlibFn func(args []values.Value, ctx ExecContext) (values.Value, error)

// stdlibFuncs is the dispatch map. Plans 04-06 register implementations here.
var stdlibFuncs = map[string]stdlibFn{}

func evalApply(e *ast.ApplyExpr, env *Env, ctx ExecContext) (values.Value, error) {
    fn, ok := stdlibFuncs[e.Name]
    if !ok {
        return nil, &EvalError{Pos: e.Pos, Message: fmt.Sprintf("unknown function %q", e.Name)}
    }
    args := make([]values.Value, len(e.Args))
    for i, a := range e.Args {
        v, err := Eval(a, env, ctx)
        if err != nil {
            return nil, err
        }
        args[i] = v
    }
    return fn(args, ctx)
}
```
