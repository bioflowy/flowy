```go
// handleImportFile handles POST /projects/{slug}/files.
// Dispatches to browser upload (multipart) or URI import (JSON) based on Content-Type (per D-01).
// Requires executor role (per D-14).
func (s *Server) handleImportFile(w http.ResponseWriter, r *http.Request) {
    proj := projectFromContext(r.Context())
    if proj == nil {
        writeError(w, http.StatusNotFound, "project not found")
        return
    }
    ct := r.Header.Get("Content-Type")
    if strings.HasPrefix(ct, "multipart/form-data") {
        s.handleImportFileBrowser(w, r, proj)
    } else {
        s.handleImportFileURI(w, r, proj)
    }
}
```

**Browser upload branch** (`handleImportFileBrowser`) — mirrors `handleUpload` from `handlers_uploads.go` exactly (per D-03):

```go
func (s *Server) handleImportFileBrowser(w http.ResponseWriter, r *http.Request, proj *store.Project) {
    r.Body = http.MaxBytesReader(w, r.Body, 100<<20)
    if err := r.ParseMultipartForm(32 << 20); err != nil {
        writeError(w, http.StatusBadRequest, "failed to parse multipart form (max 100 MB)")
        return
    }
    file, header, err := r.FormFile("file")
    if err != nil {
        writeError(w, http.StatusBadRequest, fmt.Sprintf("missing 'file' field: %v", err))
        return
    }
    defer file.Close()

    // Sanitize filename (per D-11).
    filename := filepath.Base(header.Filename)
    uploadID := uuid.New().String()
    storagePath := filepath.Join(s.cfg.BaseDir, "projects", proj.Slug, "uploads", uploadID, filename)

    // Path traversal guard (same as handleUpload).
    expectedPrefix := filepath.Clean(s.cfg.BaseDir) + string(filepath.Separator)
    if !strings.HasPrefix(storagePath, expectedPrefix) {
        writeError(w, http.StatusBadRequest, "invalid upload path")
        return
    }

    if err := s.backend.UploadFileStream(storagePath, file); err != nil {
        writeError(w, http.StatusInternalServerError, "failed to store uploaded file")
        return
    }

    // Get size after upload (per D-06 pattern).
    size, err := s.backend.Stat(storagePath)
    if err != nil {
        log.Printf("handleImportFileBrowser: stat %q after upload: %v", storagePath, err)
        size = 0
    }

    // Create managed file record (per D-04).
    now := time.Now().UTC()
    mf := &store.ManagedFile{
        ID:        uuid.New().String(),
        ProjectID: proj.ID,
        Path:      storagePath,
        FileType:  "",  // Phase 38: file_type defaults to empty string
        Label:     filename, // per D-11
        Size:      size,
        CreatedAt: now,
    }
    if err := s.store.CreateManagedFile(r.Context(), mf); err != nil {
        log.Printf("handleImportFileBrowser: create managed file: %v", err)
        writeError(w, http.StatusInternalServerError, "failed to register file")
        return
    }

    resp := managedFileToResponse(mf)
    writeJSON(w, http.StatusCreated, map[string]api.ManagedFileResponse{"file": resp})
}
```

**URI import branch** (`handleImportFileURI`) — handles both S3 and local path (per D-02):

```go
func (s *Server) handleImportFileURI(w http.ResponseWriter, r *http.Request, proj *store.Project) {
    var req api.ImportFileRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        writeError(w, http.StatusBadRequest, "invalid JSON body")
        return
    }
    if req.URI == "" {
        writeError(w, http.StatusBadRequest, "uri is required")
        return
    }

    var (
        filePath string  // stored as ManagedFile.Path
        label    string
        size     int64
    )

    if strings.HasPrefix(req.URI, "s3://") {
        // S3 URI mode (per D-05, D-06).
        _, key, err := storage.ParseS3URI(req.URI)
        if err != nil {
            writeError(w, http.StatusBadRequest, fmt.Sprintf("invalid S3 URI: %v", err))
            return
        }
        // CRITICAL: Pass key (not full URI) to Stat — S3Backend.Stat uses key as HeadObject Key.
        size, err = s.backend.Stat(key)
        if err != nil {
            writeError(w, http.StatusBadRequest, fmt.Sprintf("S3 object not accessible: %v", err))
            return
        }
        filePath = req.URI   // Store full URI for future reference
        label = filepath.Base(req.URI) // per D-11
    } else {
        // Local path mode (per D-07, D-08, D-09, D-10).
        if s.cfg.ImportRoot == "" {
            writeError(w, http.StatusBadRequest, "local path import not configured")
            return
        }
        importRoot := filepath.Clean(s.cfg.ImportRoot)
        cleanedPath := filepath.Clean(req.URI)
        // Path traversal guard (per D-09): check prefix with separator to prevent
        // "/srv/data2/..." matching "/srv/data" prefix.
        if cleanedPath != importRoot &&
            !strings.HasPrefix(cleanedPath, importRoot+string(filepath.Separator)) {
            writeError(w, http.StatusBadRequest, "path is outside import root")
            return
        }
        var err error
        size, err = s.backend.Stat(cleanedPath)
        if err != nil {
            writeError(w, http.StatusBadRequest, fmt.Sprintf("file not accessible: %v", err))
            return
        }
        filePath = cleanedPath
        label = filepath.Base(cleanedPath) // per D-11
    }

    now := time.Now().UTC()
    mf := &store.ManagedFile{
        ID:        uuid.New().String(),
        ProjectID: proj.ID,
        Path:      filePath,
        FileType:  "",  // Phase 38: file_type defaults to empty string
        Label:     label,
        Size:      size,
        CreatedAt: now,
    }
    if err := s.store.CreateManagedFile(r.Context(), mf); err != nil {
        log.Printf("handleImportFileURI: create managed file: %v", err)
        writeError(w, http.StatusInternalServerError, "failed to register file")
        return
    }

    resp := managedFileToResponse(mf)
    writeJSON(w, http.StatusCreated, map[string]api.ManagedFileResponse{"file": resp})
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client -> handleImportFile | Untrusted multipart file content and filename, or untrusted JSON URI |
| handler -> backend.Stat | User-supplied S3 key or local path passed to storage layer |
| handler -> backend.UploadFileStream | User-uploaded file content stored to server filesystem/S3 |

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-38-04 | Tampering / EoP | handleImportFileURI (local path) | mitigate | `filepath.Clean` + `strings.HasPrefix(cleanedPath, importRoot+string(filepath.Separator))` prevents ImportRoot escape (D-09); verified by TestHandleImportFile_LocalPath_Traversal |
| T-38-05 | Denial of Service | handleImportFileBrowser | mitigate | `http.MaxBytesReader(w, r.Body, 100<<20)` limits upload to 100 MB before parsing multipart form |
| T-38-06 | Information Disclosure | handleImportFileURI (S3) | accept | S3 error messages from HeadObject may reveal bucket configuration; acceptable since only executor role can access; error wrapped in generic "S3 object not accessible" message |
| T-38-07 | Spoofing | handleImportFileURI (local path) | mitigate | `ImportRoot` config restricts filesystem access scope; empty ImportRoot disables local imports entirely (D-08); verified by TestHandleImportFile_LocalPath_NoImportRoot |
