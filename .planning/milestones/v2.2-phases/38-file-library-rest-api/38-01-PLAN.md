```go
package server

import (
    "log"
    "net/http"

    "github.com/go-chi/chi/v5"

    "github.com/bioflowy/flowy/internal/api"
    "github.com/bioflowy/flowy/internal/store"
)

// managedFileToResponse converts a store.ManagedFile to an api.ManagedFileResponse.
// Omits run_id per D-16 — it is not user-facing in Phase 38.
func managedFileToResponse(f *store.ManagedFile) api.ManagedFileResponse {
    return api.ManagedFileResponse{
        ID:        f.ID,
        Path:      f.Path,
        FileType:  f.FileType,
        Label:     f.Label,
        Size:      f.Size,
        CreatedAt: f.CreatedAt.Format("2006-01-02T15:04:05Z"),
    }
}

// handleListFiles handles GET /projects/{slug}/files.
// Returns the list of managed files for the project (per D-16).
// Requires viewer role (per D-13).
func (s *Server) handleListFiles(w http.ResponseWriter, r *http.Request) {
    proj := projectFromContext(r.Context())
    if proj == nil {
        writeError(w, http.StatusNotFound, "project not found")
        return
    }
    files, err := s.store.ListManagedFiles(r.Context(), proj.ID)
    if err != nil {
        log.Printf("handleListFiles: list files for project %q: %v", proj.Slug, err)
        writeError(w, http.StatusInternalServerError, "failed to list files")
        return
    }
    resp := make([]api.ManagedFileResponse, len(files))
    for i, f := range files {
        f := f
        resp[i] = managedFileToResponse(&f)
    }
    writeJSON(w, http.StatusOK, api.ListManagedFilesResponse{Files: resp})
}

// handleDeleteFile handles DELETE /projects/{slug}/files/{id}.
// Removes the managed file record without touching storage (per D-18, FILE-02).
// Requires executor role (per D-15).
func (s *Server) handleDeleteFile(w http.ResponseWriter, r *http.Request) {
    proj := projectFromContext(r.Context())
    if proj == nil {
        writeError(w, http.StatusNotFound, "project not found")
        return
    }
    id := chi.URLParam(r, "id")
    // Fetch first to detect 404 — SQLite DELETE doesn't error on missing rows.
    mf, err := s.store.GetManagedFileByID(r.Context(), id)
    if err != nil {
        writeError(w, http.StatusNotFound, "managed file not found")
        return
    }
    // Cross-project guard: return 404 (not 403) to avoid disclosing existence.
    if mf.ProjectID != proj.ID {
        writeError(w, http.StatusNotFound, "managed file not found")
        return
    }
    if err := s.store.DeleteManagedFile(r.Context(), id); err != nil {
        log.Printf("handleDeleteFile: delete file %q: %v", id, err)
        writeError(w, http.StatusInternalServerError, "failed to delete managed file")
        return
    }
    w.WriteHeader(http.StatusNoContent)
}
```
