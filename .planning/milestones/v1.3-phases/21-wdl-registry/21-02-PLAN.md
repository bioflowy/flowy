// RegistryPushRequest is the request body for POST /projects/{slug}/registry.
type RegistryPushRequest struct {
    // Name is the entry name (slug format: lowercase alphanumeric + hyphens, 1-64 chars).
    Name string `json:"name"`

    // Content is the raw WDL file content as a string.
    Content string `json:"content"`
}

// RegistryPushResponse is the response body for POST /projects/{slug}/registry.
type RegistryPushResponse struct {
    Name      string `json:"name"`
    Digest    string `json:"digest"`
    Kind      string `json:"kind"`
    VersionID string `json:"version_id"`
}

// RegistryPullResponse is the response body for GET /projects/{slug}/registry/{name}.
type RegistryPullResponse struct {
    Name    string `json:"name"`
    Digest  string `json:"digest"`
    Kind    string `json:"kind"`
    Content string `json:"content"`
}

// RegistryEntryResponse is a single entry in the list response.
type RegistryEntryResponse struct {
    Name         string   `json:"name"`
    Kind         string   `json:"kind"`
    VersionCount int      `json:"version_count"`
    Tags         []string `json:"tags"`
    CreatedAt    string   `json:"created_at"`
}

// RegistryVersionResponse is a single version in the versions list response.
type RegistryVersionResponse struct {
    ID        string   `json:"id"`
    Digest    string   `json:"digest"`
    Tags      []string `json:"tags"`
    CreatedAt string   `json:"created_at"`
}

// RegistryTagRequest is the request body for POST /projects/{slug}/registry/{name}/tags.
type RegistryTagRequest struct {
    // Tag is the tag name (any string up to 128 chars, no '@' or ':').
    Tag string `json:"tag"`

    // Digest is the version digest to tag ("sha256:<hex>").
    Digest string `json:"digest"`
}
```

func computeDigest(content string) string {
    h := sha256.Sum256([]byte(content))
    return fmt.Sprintf("sha256:%x", h)
}
```

**classifyWDL helper:**
```go
func classifyWDL(name, content string) (string, error) {
    doc, err := parser.Parse(name, []byte(content))
    if err != nil {
        return "", fmt.Errorf("WDL parse error: %w", err)
    }
    if len(doc.Workflows) > 0 {
        return "workflow", nil
    }
    if len(doc.Tasks) > 0 {
        return "tool", nil
    }
    return "", fmt.Errorf("WDL file defines neither a task nor a workflow")
}
```

**validTagName helper:**
```go
func validTagName(s string) bool {
    return len(s) >= 1 && len(s) <= 128 && !strings.ContainsAny(s, "@:")
}
```

**handleRegistryPush** (POST /projects/{slug}/registry):
- Extract project from context via `projectFromContext(r.Context())`
- Decode `api.RegistryPushRequest` from body
- Validate `req.Name` with `validSlug(req.Name)` — return 400 if invalid: `"invalid entry name: must be lowercase alphanumeric with hyphens, 1-64 chars"`
- Validate `req.Content` is not empty — return 400: `"content is required"`
- Call `classifyWDL(req.Name, req.Content)` — return 422 on error with the error message
- Call `computeDigest(req.Content)` to get digest string
- Call `s.store.PushRegistryVersion(r.Context(), proj.ID, req.Name, kind, digest, req.Content)`
- If error contains "kind" (kind mismatch), return 409 Conflict with the error message
- If other error, return 500
- If `created` is true, write 201; if false, write 200
- Response body: `api.RegistryPushResponse{Name: req.Name, Digest: version.Digest, Kind: kind, VersionID: version.ID}`

**handleRegistryPull** (GET /projects/{slug}/registry/{name}):
- Extract project from context
- Get `name` from `chi.URLParam(r, "name")`
- Get `tag` from `r.URL.Query().Get("tag")`
- Get `digest` from `r.URL.Query().Get("digest")`
- If both empty, return 400: `"must specify ?tag= or ?digest= query parameter"`
- If both non-empty, return 400: `"specify either ?tag= or ?digest=, not both"`
- If tag: call `s.store.GetRegistryVersionByTag(r.Context(), proj.ID, name, tag)` — on error return 404: `fmt.Sprintf("tag %q not found for entry %q", tag, name)`
- If digest: call `s.store.GetRegistryVersionByDigest(r.Context(), proj.ID, name, digest)` — on error return 404: `fmt.Sprintf("digest %q not found for entry %q", digest, name)`
- Response 200: `api.RegistryPullResponse{Name: entry.Name, Digest: version.Digest, Kind: entry.Kind, Content: version.WDLContent}`
