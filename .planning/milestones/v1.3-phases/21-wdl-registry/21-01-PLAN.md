-- +goose Up
-- Creates registry tables for WDL content-addressed registry support.
-- registry_entries: named slots with kind classification (tool/workflow).
-- registry_versions: immutable content-addressed WDL blobs.
-- registry_tags: mutable name pointers to specific versions.

CREATE TABLE IF NOT EXISTS registry_entries (
    id         TEXT     PRIMARY KEY,
    project_id TEXT     NOT NULL REFERENCES projects(id),
    name       TEXT     NOT NULL,
    kind       TEXT     NOT NULL,
    is_active  INTEGER  NOT NULL DEFAULT 1,
    created_at DATETIME NOT NULL DEFAULT (datetime('now')),
    updated_at DATETIME NOT NULL DEFAULT (datetime('now')),
    UNIQUE (project_id, name)
);
CREATE INDEX IF NOT EXISTS idx_registry_entries_project ON registry_entries(project_id);

CREATE TABLE IF NOT EXISTS registry_versions (
    id          TEXT     PRIMARY KEY,
    entry_id    TEXT     NOT NULL REFERENCES registry_entries(id),
    digest      TEXT     NOT NULL,
    wdl_content TEXT     NOT NULL,
    created_at  DATETIME NOT NULL DEFAULT (datetime('now')),
    UNIQUE (entry_id, digest)
);
CREATE INDEX IF NOT EXISTS idx_registry_versions_entry ON registry_versions(entry_id);
CREATE INDEX IF NOT EXISTS idx_registry_versions_digest ON registry_versions(digest);

CREATE TABLE IF NOT EXISTS registry_tags (
    id         TEXT     PRIMARY KEY,
    entry_id   TEXT     NOT NULL REFERENCES registry_entries(id),
    name       TEXT     NOT NULL,
    version_id TEXT     NOT NULL REFERENCES registry_versions(id),
    created_at DATETIME NOT NULL DEFAULT (datetime('now')),
    updated_at DATETIME NOT NULL DEFAULT (datetime('now')),
    UNIQUE (entry_id, name)
);
CREATE INDEX IF NOT EXISTS idx_registry_tags_entry ON registry_tags(entry_id);

-- +goose Down
DROP TABLE IF EXISTS registry_tags;
DROP TABLE IF EXISTS registry_versions;
DROP TABLE IF EXISTS registry_entries;
```

// RegistryEntry represents a named WDL entry in a project registry.
// Kind is "tool" (WDL task) or "workflow" (WDL workflow), set on first push and immutable.
// IsActive is false for soft-deleted entries.
type RegistryEntry struct {
    ID        string
    ProjectID string
    Name      string
    Kind      string // "tool" or "workflow"
    IsActive  bool
    CreatedAt time.Time
    UpdatedAt time.Time
}

// RegistryVersion represents an immutable content-addressed WDL blob.
// Digest is "sha256:<hex>" computed from WDLContent.
type RegistryVersion struct {
    ID         string
    EntryID    string
    Digest     string // "sha256:<hex>"
    WDLContent string
    CreatedAt  time.Time
}

// RegistryTag is a mutable pointer from a human-readable tag name to a specific version.
type RegistryTag struct {
    ID        string
    EntryID   string
    Name      string
    VersionID string
    CreatedAt time.Time
    UpdatedAt time.Time
}

// RegistryEntryListItem is the summary row returned by ListRegistryEntries.
type RegistryEntryListItem struct {
    Name         string
    Kind         string
    VersionCount int
    Tags         []string // tag names only
    CreatedAt    time.Time
}

// RegistryVersionListItem is a version row with associated tag names.
type RegistryVersionListItem struct {
    ID        string
    Digest    string
    Tags      []string
    CreatedAt time.Time
}
```

**PushRegistryVersion** — the most complex method. Must use writer connection for both SELECT and INSERT (TOCTOU avoidance):
1. `s.writer.QueryRowContext` to SELECT existing entry by `(project_id, name, is_active=1)`
2. If entry exists and `entry.Kind != kind` parameter, return `fmt.Errorf("sqlite: entry %q is kind %q; cannot push kind %q", name, entry.Kind, kind)` — caller converts to 409
3. If entry does not exist, INSERT new entry with `uuid.New().String()` as ID, `is_active=1`
4. Then SELECT existing version by `(entry_id, digest)` on writer
5. If version exists, return `(existingVersion, false, nil)` — idempotent
6. If version does not exist, INSERT new version with `uuid.New().String()` as ID
7. Return `(newVersion, true, nil)`
8. Use `time.Now().UTC()` for timestamps, format as `time.RFC3339`

**GetRegistryEntry** — simple reader query:
```go
func (s *SQLiteStore) GetRegistryEntry(ctx context.Context, projectID, name string) (*store.RegistryEntry, error) {
    row := s.reader.QueryRowContext(ctx,
        `SELECT id, project_id, name, kind, is_active, created_at, updated_at
         FROM registry_entries WHERE project_id = ? AND name = ? AND is_active = 1`, projectID, name)
    return scanRegistryEntry(row)
}
```

**GetRegistryVersionByTag** — reader query with JOIN across all three tables:
```sql
SELECT rv.id, rv.entry_id, rv.digest, rv.wdl_content, rv.created_at,
       re.id, re.project_id, re.name, re.kind, re.is_active, re.created_at, re.updated_at
FROM registry_tags rt
JOIN registry_versions rv ON rv.id = rt.version_id
JOIN registry_entries re ON re.id = rt.entry_id
WHERE re.project_id = ? AND re.name = ? AND rt.name = ? AND re.is_active = 1
```

**GetRegistryVersionByDigest** — reader query with JOIN:
```sql
SELECT rv.id, rv.entry_id, rv.digest, rv.wdl_content, rv.created_at,
       re.id, re.project_id, re.name, re.kind, re.is_active, re.created_at, re.updated_at
FROM registry_versions rv
JOIN registry_entries re ON re.id = rv.entry_id
WHERE re.project_id = ? AND re.name = ? AND rv.digest = ? AND re.is_active = 1
```

**ListRegistryEntries** — reader query. Use a two-query approach for simplicity (acceptable at small scale):
1. First query: SELECT entries with version count via subquery:
```sql
SELECT re.name, re.kind, re.created_at,
       (SELECT COUNT(*) FROM registry_versions rv WHERE rv.entry_id = re.id) as version_count
FROM registry_entries re
WHERE re.project_id = ? AND re.is_active = 1
ORDER BY re.created_at ASC
```
2. Second query per entry: SELECT tag names for each entry:
```sql
SELECT rt.name FROM registry_tags rt
JOIN registry_entries re ON re.id = rt.entry_id
WHERE re.project_id = ? AND re.name = ? AND re.is_active = 1
ORDER BY rt.name ASC
```
Note: This is an N+1 query. Add a comment: `// TODO: optimize with a single JOIN query if entry count grows large`

**ListRegistryVersions** — reader query:
1. First get entry by `(projectID, name, is_active=1)` — return error if not found
2. Query versions:
```sql
SELECT id, digest, created_at FROM registry_versions WHERE entry_id = ? ORDER BY created_at ASC
```
3. For each version, query tags:
```sql
SELECT name FROM registry_tags WHERE version_id = ? ORDER BY name ASC
```
Return `[]store.RegistryVersionListItem` with `ID`, `Digest`, `Tags`, `CreatedAt`.
